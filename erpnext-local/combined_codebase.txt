def initialize_dong_phong_defaults():
    """
    Automates prerequisite master data initialization for Dong Phong Logistics.
    Safe to run multiple times (idempotent).
    """
    frappe.set_user("Administrator")
    print("🚀 Starting Dong Phong Logistics Data Initialization...")

    # 1. Ensure UOM "Litre" exists
    if not frappe.db.exists("UOM", "Litre"):
        uom = frappe.get_doc({
            "doctype": "UOM",
            "uom_name": "Litre",
            "must_be_whole_number": 0
        })
        uom.insert(ignore_permissions=True)
        print("  [+] Created UOM: Litre")
    else:
        print("  [=] UOM Litre already exists")

    # 2. Ensure Customer Group "Commercial" exists
    if not frappe.db.exists("Customer Group", "Commercial"):
        cg = frappe.get_doc({
            "doctype": "Customer Group",
            "customer_group_name": "Commercial",
            "is_group": 0
        })
        cg.insert(ignore_permissions=True)
        print("  [+] Created Customer Group: Commercial")
    else:
        print("  [=] Customer Group Commercial already exists")

    # 3. Ensure Territory "Vietnam" exists
    if not frappe.db.exists("Territory", "Vietnam"):
        ter = frappe.get_doc({
            "doctype": "Territory",
            "territory_name": "Vietnam",
            "is_group": 0
        })
        ter.insert(ignore_permissions=True)
        print("  [+] Created Territory: Vietnam")
    else:
        print("  [=] Territory Vietnam already exists")

    # 4. Create or Update Vehicle 51C-123.45
    vehicle_license = "51C-123.45"
    if not frappe.db.exists("Vehicle", vehicle_license):
        vehicle = frappe.get_doc({
            "doctype": "Vehicle",
            "license_plate": vehicle_license,
            "make": "Dong Phong Demo",
            "model": "Truck 2026",
            "payload_capacity": 2000,
            "custom_technical_status": "Available" if frappe.db.has_column("Vehicle", "custom_technical_status") else None
        })
        vehicle.insert(ignore_permissions=True)

        vehicle.payload_capacity = 2000

            vehicle.custom_technical_status = "Available"

    frappe.db.commit()

from frappe.utils import add_days, today

def run():
    results=[]
    original_counts={dt:frappe.db.count(dt) for dt in ['Vehicle','Driver','Delivery Trip']}
    original_user=frappe.session.user
    frappe.set_user('Administrator')
    frappe.db.savepoint('dispatch_suite')
    try:
        vehicles=[];drivers=[]
        for i in range(3):
            v=frappe.get_doc(dict(doctype='Vehicle',license_plate='DP-TEST-'+frappe.generate_hash(length=8),
                make='Simulation',model='Test',last_odometer=1000,fuel_type='Diesel',uom='Litre',
                custom_payload_capacity_kg=2000,custom_technical_status='Available')).insert()
            vehicles.append(v.name)
            drivers.append(frappe.get_doc(dict(doctype='Driver',full_name='DP TEST '+str(i),status='Active',naming_series='HR-DRI-.YYYY.-')).insert().name)
        template=frappe.get_doc('Delivery Trip','MAT-DT-2026-00001')
        date=add_days(today(),10)
        def trip(vi=0,di=0,start='08:00:00',end='12:00:00',load=1000):
            return frappe.get_doc(dict(doctype='Delivery Trip',naming_series='MAT-DT-.YYYY.-',company=template.company,
                vehicle=vehicles[vi],driver=drivers[di],departure_time=date+' '+start,
                custom_expected_end_time=date+' '+end,custom_planned_load_kg=load,
                custom_start_odometer=1000,
                delivery_stops=[dict(customer=s.customer,address=s.address) for s in template.delivery_stops]))
        def reject(label,code,fn):
            frappe.db.savepoint('rejected_case')
            try: fn()
            except frappe.ValidationError as e:
                if '['+code+']' not in str(e): raise
                results.append(label)
            else: raise AssertionError('Expected rejection: '+label)
            finally: frappe.db.rollback(save_point='rejected_case')
        valid=trip().insert();valid.submit()
        assert valid.docstatus==1 and valid.status=='Scheduled'
        results.append('Valid trip inserts and submits')
        reject('Same vehicle overlap blocked','DP_CONFLICT',lambda:trip(di=1).insert())
        reject('Same driver overlap blocked','DP_CONFLICT',lambda:trip(vi=1).insert())
        other=trip(vi=1,di=1).insert();other.submit();results.append('Different resources may overlap')
        boundary=trip(start='12:00:00',end='14:00:00').insert();boundary.submit();results.append('Adjacent intervals accepted')
        reject('Reversed time rejected','DP_TIME',lambda:trip(vi=2,di=2,start='12:00:00',end='08:00:00').insert())
        reject('Equal timestamps rejected','DP_TIME',lambda:trip(vi=2,di=2,start='08:00:00',end='08:00:00').insert())
        reject('Overload rejected','DP_OVERLOAD',lambda:trip(vi=2,di=2,load=2001).insert())
        def no_end():
            d=trip(vi=2,di=2);d.custom_expected_end_time=None;d.insert();d.submit()
        reject('Missing end cannot submit','DP_REQUIRED',no_end)
        reject('Missing load cannot submit','DP_LOAD_REQUIRED',lambda:trip(vi=2,di=2,load=0).insert().submit())
        def no_capacity():
            frappe.db.set_value('Vehicle',vehicles[2],'custom_payload_capacity_kg',0)
            trip(vi=2,di=2).insert().submit()
        reject('Unknown capacity cannot submit','DP_LOAD_REQUIRED',no_capacity)
        def unavailable():
            frappe.db.set_value('Vehicle',vehicles[2],'custom_technical_status','Under Maintenance')
            trip(vi=2,di=2).insert()
        reject('Maintenance vehicle rejected','DP_UNAVAILABLE',unavailable)
        def inactive():
            frappe.db.set_value('Driver',drivers[2],'status','Suspended')
            trip(vi=2,di=2).insert()
        reject('Inactive driver rejected','DP_DRIVER',inactive)
        def expired():
            frappe.db.set_value('Driver',drivers[2],'expiry_date',add_days(date,-1))
            trip(vi=2,di=2).insert()
        reject('Expired licence rejected','DP_LICENSE',expired)
        def km_bad():
            d=trip(vi=2,di=2);d.custom_end_odometer=999;d.insert()
        reject('Odometer regression rejected','DP_ODOMETER',km_bad)
        def km_after_submit():
            d=frappe.get_doc('Delivery Trip',valid.name);d.custom_end_odometer=999;d.save()
        reject('Odometer checked after submit','DP_ODOMETER',km_after_submit)
        draft1=trip(vi=2,di=2).insert();draft2=trip(vi=2,di=2).insert()
        results.append('Drafts do not reserve resources')
        draft1.submit()
        reject('Submit rechecks conflict','DP_CONFLICT',lambda:draft2.submit())
        valid.cancel()
        replacement=trip().insert();replacement.submit();results.append('Cancellation releases reservation')
        for stop in replacement.delivery_stops: stop.visited=1
        replacement.save();assert replacement.status=='Completed'
        trip().insert().submit();results.append('Completed trip releases reservation')
    finally:
        frappe.db.rollback(save_point='dispatch_suite')
        frappe.set_user(original_user)
    assert original_counts=={dt:frappe.db.count(dt) for dt in original_counts}
    result={'passed':len(results),'checks':results,'test_data_rolled_back':True}
    print(json.dumps(result,ensure_ascii=False))
    return result

run()

"""Server-side rules. Submitted, unfinished trips reserve vehicle and driver.

Drafts are plans, not reservations. Intervals use [start, end).
Master row locks serialize submissions sharing a vehicle or driver.
"""
import math
import frappe
from frappe.utils import flt, get_datetime, getdate

def fail(code, message):
    frappe.throw(f'[{code}] {message}')

def number(value, label):
    n=flt(value)
    if not math.isfinite(n) or n<0:
        fail('DP_NUMBER',label+' phải là số không âm hợp lệ.')
    return n

def validate_vehicle(doc, method=None):
    number(doc.get('custom_payload_capacity_kg'),'Tải trọng xe')

def validate_odometer(doc, method=None):
    start=number(doc.get('custom_start_odometer'),'Số km đầu chuyến')
    end=number(doc.get('custom_end_odometer'),'Số km cuối chuyến')
    if end and end<start:
        fail('DP_ODOMETER','Số km cuối chuyến phải lớn hơn hoặc bằng số km đầu chuyến.')

def validate_trip(doc, method=None):
    if doc.docstatus==2: return
    validate_odometer(doc)
    load=number(doc.get('custom_planned_load_kg'),'Khối lượng dự kiến')
    start=get_datetime(doc.departure_time) if doc.departure_time else None
    end=get_datetime(doc.custom_expected_end_time) if doc.get('custom_expected_end_time') else None

from frappe.utils import get_datetime

DRIVER_ROLE = "DP Driver"
SIM_ROLE = "DP GPS Simulator"

# Vai trò điều phối được xem lịch sử GPS (dùng cho trang Fleet Map).
# Muốn CHỈ tài xế tự xem được dữ liệu của mình: đặt DISPATCHER_ROLES = set()
DISPATCHER_ROLES = {"System Manager", "Fleet Manager"}


# ----------------------------------------------------------------------
# Helpers: xác định user là ai, có được phép xem chuyến/xe nào
# ----------------------------------------------------------------------
def _roles():
    return set(frappe.get_roles(frappe.session.user))


def _get_current_driver():
    """Trả về tên Driver liên kết với user đang đăng nhập, hoặc None."""
    user = frappe.session.user
    if user in ("Administrator", "Guest"):
        return None

    meta = frappe.get_meta("Driver")
    for field in ("custom_user", "user"):
        if meta.has_field(field):
            name = frappe.db.get_value("Driver", {field: user}, "name")
            if name:
                return name

    # Fallback: Driver -> Employee -> User
    if meta.has_field("employee"):
        emp = frappe.db.get_value("Employee", {"user_id": user}, "name")
        if emp:
            return frappe.db.get_value("Driver", {"employee": emp}, "name")
    return None


def _is_dispatcher():
    return bool(_roles() & DISPATCHER_ROLES)


def _is_simulator():
    return SIM_ROLE in _roles()


def _driver_owns_trip(driver, trip):
    return bool(driver) and frappe.db.get_value("Delivery Trip", trip, "driver") == driver


def _driver_owns_vehicle(driver, vehicle):
    if not driver:
        return False
    return bool(frappe.db.exists(
        "Delivery Trip", {"driver": driver, "vehicle": vehicle, "docstatus": ["<", 2]}
    ))

@frappe.whitelist(methods=["POST"])
def log_location():
    if frappe.session.user == "Guest":
        frappe.throw(_("Cần đăng nhập hoặc cung cấp API Key để thực hiện"), frappe.PermissionError)

    data = frappe.local.form_dict
    vehicle = data.get("vehicle")
    delivery_trip = data.get("delivery_trip")
    recorded_at = data.get("recorded_at")
    latitude = data.get("latitude")
    longitude = data.get("longitude")
    source_event_id = data.get("source_event_id")
    source_type = data.get("source_type", "Simulated")

    if not vehicle or not recorded_at or latitude is None or longitude is None or not source_event_id:
        frappe.throw(_("Thiếu thông tin bắt buộc trong bản tin GPS"))

    try:
        lat = float(latitude)
        lng = float(longitude)
        if not (-90 <= lat <= 90 and -180 <= lng <= 180):
            raise ValueError()
    except ValueError:
        frappe.throw(_("Tọa độ vĩ độ/kinh độ không hợp lệ"))

    if not frappe.db.exists("Vehicle", vehicle):
        frappe.throw(_("Xe {0} không tồn tại").format(vehicle))

    # Tài xế phải gửi theo đúng chuyến được giao, không chỉ theo chiếc xe từng lái.
    if not (_is_simulator() or _is_dispatcher()):
        if not delivery_trip or not _driver_owns_trip(_get_current_driver(), delivery_trip):
            frappe.throw(_("Bạn chỉ được ghi GPS cho chuyến được phân công."), frappe.PermissionError)

    if delivery_trip:
        if not frappe.db.exists("Delivery Trip", delivery_trip):
            frappe.throw(_("Chuyến {0} không tồn tại").format(delivery_trip))
        trip_doc = frappe.get_doc("Delivery Trip", delivery_trip)
        if trip_doc.vehicle != vehicle:
            frappe.throw(_("Xe {0} không thuộc chuyến {1}").format(vehicle, delivery_trip))

    existing = frappe.db.get_value(
        "GPS Log", {"source_event_id": source_event_id}, ["name", "latitude", "longitude"], as_dict=True
    )
    if existing:
        if float(existing.latitude) == lat and float(existing.longitude) == lng:
            return {"status": "ignored", "message": "Bản tin trùng lặp đã được xử lý", "source_event_id": source_event_id}
        else:
            frappe.throw(
                _("Xung đột: source_event_id {0} đã tồn tại với nội dung khác").format(source_event_id),
                frappe.DuplicateEntryError,
            )

    gps_doc = frappe.get_doc({
        "doctype": "GPS Log",
        "vehicle": vehicle,
        "delivery_trip": delivery_trip,
        "recorded_at": recorded_at,
        "latitude": lat,
        "longitude": lng,
        "source_event_id": source_event_id,
        "source_type": source_type,
    })
    gps_doc.insert(ignore_permissions=True)

    current_last_time = frappe.db.get_value("Vehicle", vehicle, "custom_last_gps_time")
    new_time = get_datetime(recorded_at)

    if not current_last_time or new_time >= get_datetime(current_last_time):
        frappe.db.set_value("Vehicle", vehicle, {
            "custom_last_lat": lat,
            "custom_last_lng": lng,
            "custom_last_gps_time": recorded_at,
        }, update_modified=False)

    frappe.db.commit()
    return {"status": "success", "source_event_id": source_event_id, "name": gps_doc.name}

@frappe.whitelist(methods=["GET", "POST"])
def get_gps_history(vehicle=None, delivery_trip=None, limit=100):
    if frappe.session.user == "Guest":
        frappe.throw(_("Cần đăng nhập"), frappe.PermissionError)

    if not vehicle and not delivery_trip:
        frappe.throw(_("Cần cung cấp vehicle hoặc delivery_trip"))

    driver = None
    if not _is_dispatcher():
        driver = _get_current_driver()
        if not driver:
            frappe.throw(_("Bạn không có quyền xem dữ liệu GPS"), frappe.PermissionError)
        if vehicle and not _driver_owns_vehicle(driver, vehicle):
            frappe.throw(_("Bạn không có quyền xem GPS của xe {0}").format(vehicle), frappe.PermissionError)
        if delivery_trip and not _driver_owns_trip(driver, delivery_trip):
            frappe.throw(_("Chuyến {0} không thuộc về bạn").format(delivery_trip), frappe.PermissionError)

    conditions = []
    values = {}

    if driver:
        conditions.append("delivery_trip IN (SELECT name FROM `tabDelivery Trip` WHERE driver = %(driver)s)")
        values["driver"] = driver

    if vehicle:
        conditions.append("vehicle = %(vehicle)s")
        values["vehicle"] = vehicle
    if delivery_trip:
        conditions.append("delivery_trip = %(delivery_trip)s")
        values["delivery_trip"] = delivery_trip

    where_clause = " AND ".join(conditions)
    values["limit"] = int(limit)

    query = f"""
        SELECT recorded_at, latitude, longitude, source_type, source_event_id
        FROM `tabGPS Log`
        WHERE {where_clause}
        ORDER BY recorded_at ASC
        LIMIT %(limit)s
    """

    return frappe.db.sql(query, values, as_dict=True)

@frappe.whitelist(methods=["GET", "POST"])
def get_driver_schedule():
    if frappe.session.user == "Guest":

    driver = _get_current_driver()
    
    # Strictly enforce driver identity and role check
    if not driver or DRIVER_ROLE not in _roles():

    # Removed ignore_permissions=True to respect User Permissions
    trips = frappe.get_all(
        "Delivery Trip",

            "driver": driver,
            "docstatus": ["<", 2],
            "status": ["not in", ["Completed", "Cancelled"]],
        },
        fields=["name", "vehicle", "driver_name", "status"],
        order_by="departure_time desc",
        limit_page_length=1
    )

    if not trips:
        return {"trip": None, "stops": []}

    trip = trips[0]
    stops = frappe.get_all(
        "Delivery Stop",
        filters={"parent": trip["name"], "parenttype": "Delivery Trip"},
        fields=["name", "idx", "customer", "address", "lat", "lng", "visited"],
        order_by="idx asc",
        limit_page_length=100
    )

    return {"trip": trip, "stops": stops}


@frappe.whitelist(methods=["POST"])
def complete_delivery_stop(stop_name=None):
    if frappe.session.user == "Guest":
        frappe.throw(_("Cần đăng nhập để thực hiện thao tác này"), frappe.PermissionError)

    if not stop_name:
        stop_name = frappe.local.form_dict.get("stop_name")

    if not stop_name or not frappe.db.exists("Delivery Stop", stop_name):
        frappe.throw(_("Không tìm thấy điểm dừng hợp lệ"))

    parent_trip = frappe.db.get_value("Delivery Stop", stop_name, "parent")

    # Giữ khóa chuyến đến khi cập nhật xong, tránh xác nhận đồng thời với thao tác hủy.
    trip = frappe.db.get_value(
        "Delivery Trip", parent_trip, ["driver", "docstatus", "status"],
        as_dict=True, for_update=True,
    )

    # Chỉ đúng tài xế của chuyến này mới được đánh dấu (không có ngoại lệ, kể cả quản trị)
    driver = _get_current_driver()
    if not trip or not driver or trip.driver != driver:
        frappe.throw(_("Điểm dừng này không thuộc chuyến của bạn"), frappe.PermissionError)

    if trip.docstatus == 2 or trip.status == "Cancelled":
        frappe.throw(_("Không thể xác nhận điểm giao của chuyến đã hủy."))

    frappe.db.set_value("Delivery Stop", stop_name, "visited", 1, update_modified=True)

    unvisited_count = frappe.db.count("Delivery Stop", filters={"parent": parent_trip, "visited": 0})

    trip_completed = False
    if unvisited_count == 0:
        frappe.db.set_value("Delivery Trip", parent_trip, "status", "Completed", update_modified=True)
        trip_completed = True

    frappe.db.commit()
    return {"status": "success", "stop_name": stop_name, "trip_completed": trip_completed}

    function ageMin(t) {
      if (!t) return Infinity;
      const now = moment(frappe.datetime.now_datetime().toString().slice(0, 19)); 
      return now.diff(moment(String(t).slice(0, 19)), 'minutes');
    }

    function formatTimeAgo(time) {
      const m = ageMin(time);
      if (!isFinite(m)) return 'Mất tín hiệu';
      if (m < 60) {
        return `${m} phút trước`;
      }
      const hrs = Math.floor(m / 60);
      const mins = m % 60;
      return `${hrs} giờ ${mins} phút trước`;
    }

    function getStatusColor(status, time) {
      const st = String(status || '').toLowerCase();
      if (st.includes('bảo dưỡng') || st.includes('sửa chữa') || st.includes('maintenance')) return '#ea580c';
      if (st.includes('hỏng') || st.includes('sự cố') || st.includes('out of service') || st.includes('breakdown')) return '#dc2626';
      const m = ageMin(time);
      return m < 5 ? '#16a34a' : '#000000';
    }

      return isAvailable || isOnline;
    }

    function loadVehicles(isRefresh) {
      return frappe.call({
        method: 'frappe.client.get_list',
        args: {
          doctype: 'Vehicle',
          fields: ['name', 'custom_technical_status', 'custom_gps_device_id', 'custom_last_lat', 'custom_last_lng', 'custom_last_gps_time'],
          order_by: 'name asc',
          limit_page_length: 500
        },
        callback: function(r) {
          if (!r.message) return;
          let online = 0;
          vehiclesData = [];

          const promises = r.message.map(function(row) {
            if (row.custom_last_lat && row.custom_last_lng) {
              const vObj = {
                plate: row.name,
                status: row.custom_technical_status,
                device: row.custom_gps_device_id,
                lat: row.custom_last_lat,
                lng: row.custom_last_lng,
                time: row.custom_last_gps_time,
                source_type: 'Hardware GPS'
              };

              return frappe.db.get_value('GPS Log', { vehicle: row.name }, 'source_type')
                .then(function(res) {
                  if (res && res.message && res.message.source_type) {
                    vObj.source_type = res.message.source_type;
                  }
                  vehiclesData.push(vObj);
                  upsertVehicle(vObj);
                  if (ageMin(row.custom_last_gps_time) < 5) online++;
                });
            }
            return Promise.resolve();
          });

          Promise.all(promises).then(function() {
            $('#stat-total').text(r.message.length + ' xe');

              selectVehicle(selectedPlate, true);
            }
          });
        }
      });
    }

    function calcDist(lat1, lon1, lat2, lon2) {
      const R = 6371;
      const dLat = (lat2 - lat1) * Math.PI / 180;
      const dLon = (lon2 - lon1) * Math.PI / 180;
      const a = Math.sin(dLat/2) * Math.sin(dLat/2) +
                Math.cos(lat1 * Math.PI / 180) * Math.cos(lat2 * Math.PI / 180) * 
                Math.sin(dLon/2) * Math.sin(dLon/2);
      return R * 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a));
    }

    function optimizeRoute(startLat, startLng, stops) {
      let unvisited = stops.slice();
      let currentLat = startLat;
      let currentLng = startLng;
      let orderedRoute = [];

      while (unvisited.length > 0) {
        let nearestIdx = 0;
        let minD = Infinity;

        for (let i = 0; i < unvisited.length; i++) {
          let d = calcDist(currentLat, currentLng, unvisited[i].lat, unvisited[i].lng);
          if (d < minD) {
            minD = d;
            nearestIdx = i;
          }
        }

        let nextStop = unvisited.splice(nearestIdx, 1)[0];
        currentLat = nextStop.lat;
        currentLng = nextStop.lng;
        orderedRoute.push(nextStop);
      }
      return orderedRoute;
    }

    function drawRouteFromTruck(plate, truckLat, truckLng) {
      routeLayer.clearLayers();
      stopMarkersList.length = 0;

      frappe.db.get_list('Delivery Trip', {
        filters: { vehicle: plate, docstatus: ['<', 2] },
        fields: ['name', 'driver_name', 'departure_time'],
        order_by: 'creation desc',
        limit: 1
      }).then(function(trips) {
        if (!trips || !trips.length) {
          if (markers[plate]) {
            map.setView([truckLat, truckLng], 15);
          }
          return frappe.show_alert({ message: __('Xe {0} chưa có chuyến giao hàng active', [plate]), indicator: 'orange' });
        }

        const trip = trips[0];
        frappe.db.get_list('Delivery Stop', {
          parent_doctype: 'Delivery Trip',
          filters: { parent: trip.name },
          fields: ['idx', 'customer', 'lat', 'lng', 'address'],
          limit: 100
        }).then(function(stops) {
          const validStops = stops.filter(s => s.lat && s.lng);
          if (!validStops.length) {
            if (markers[plate]) {
              map.setView([truckLat, truckLng], 15);
            }
            return frappe.show_alert({ message: __('Chuyến {0} chưa có tọa độ điểm giao hàng', [trip.name]), indicator: 'orange' });
          }

          const sortedStops = optimizeRoute(truckLat, truckLng, validStops);
          const path = [[truckLat, truckLng]].concat(sortedStops.map(s => [s.lat, s.lng]));
          const polyline = L.polyline(path, { color: '#18181b', weight: 4, dashArray: '6, 6', opacity: 0.85 }).addTo(routeLayer);

          sortedStops.forEach(function(s, seq) {
            const seqNum = seq + 1;
            const sm = L.marker([s.lat, s.lng], { icon: createStopIcon(seqNum) }).addTo(routeLayer)
              .bindPopup(`
                <div style="padding: 4px; min-width: 180px;">
                  <b style="color: #ffffff; font-size: 13px;">
                    <i class="fa fa-map-marker" style="color: #ffffff; margin-right: 6px;"></i>Điểm ghé #${seqNum}
                  </b>
                  <div style="font-size: 11px; color: #d4d4d8; margin-top: 4px;"> 
                    <div>Khách hàng: <b>${esc(s.customer || 'N/A')}</b></div>
                    <div>Địa chỉ: ${esc(s.address || 'N/A')}</div>
                  </div>
                </div>
              `);
            stopMarkersList.push(sm);
          });

          map.fitBounds(polyline.getBounds(), { padding: [80, 80], maxZoom: 14 });
        });
      });
    }

dp_gps.load_leaflet = function () {
  const CSS = 'https://unpkg.com/leaflet@1.9.4/dist/leaflet.css';
  const JS = 'https://unpkg.com/leaflet@1.9.4/dist/leaflet.js';
  if (!document.querySelector('link[href="' + CSS + '"]')) {
    const l = document.createElement('link'); l.rel = 'stylesheet'; l.href = CSS; document.head.appendChild(l);
  }
  return new Promise((resolve, reject) => {
    if (window.L) return resolve();
    const s = document.createElement('script'); s.src = JS; s.onload = resolve; s.onerror = reject;
    document.head.appendChild(s);
  });
};

dp_gps.show_trip_map = async function (frm) {
  try { await dp_gps.load_leaflet(); } catch (e) { return frappe.msgprint(__('Cannot load Leaflet')); }
  const esc = frappe.utils.escape_html;
  const stops = (frm.doc.delivery_stops || []).filter(s => s.lat && s.lng);
  const hasOrigin = frm.doc.custom_origin_lat && frm.doc.custom_origin_lng;
  const track = await frappe.db.get_list('GPS Location Log', {
    filters: { delivery_trip: frm.doc.name }, fields: ['latitude', 'longitude'],
    order_by: 'recorded_at asc', limit: 3000 });
  let veh = null;
  if (frm.doc.vehicle) {
    const r = await frappe.db.get_value('Vehicle', frm.doc.vehicle,
      ['custom_last_lat', 'custom_last_lng', 'custom_last_gps_time']);
    veh = r.message;
  }
  if (!stops.length && !track.length) return frappe.msgprint(__('This trip has no coordinates yet'));

  const d = new frappe.ui.Dialog({
    title: __('Trip map: {0}', [frm.doc.name]), size: 'extra-large',
    fields: [{ fieldtype: 'HTML', fieldname: 'map_html' }] });
  d.fields_dict.map_html.$wrapper.html(
    '<div id="dp-trip-map" style="height:60vh;min-height:360px;border-radius:8px"></div>' +
    '<div class="text-muted small" style="margin-top:6px">' +
    __('Dashed red: planned route') + ' · ' + __('Blue: GPS track') + ' · ' + __('Green dot: last vehicle position') +
    '</div>');
  d.show();

  setTimeout(() => {
    const map = L.map('dp-trip-map');
    L.tileLayer('https://tile.openstreetmap.org/{z}/{x}/{y}.png',
      { maxZoom: 19, attribution: '&copy; OpenStreetMap contributors' }).addTo(map);
    const all = [];
    const o = hasOrigin ? [frm.doc.custom_origin_lat, frm.doc.custom_origin_lng] : null;
    const path = (o ? [o] : []).concat(stops.map(s => [s.lat, s.lng]), o ? [o] : []);
    if (path.length > 1) {
      L.polyline(path, { color: '#dc2626', weight: 3, dashArray: '6 6' }).addTo(map);
      path.forEach(p => all.push(p));
    }
    if (o) L.marker(o).addTo(map).bindPopup(__('Depot'));
    stops.forEach(s => L.marker([s.lat, s.lng]).addTo(map).bindPopup(
      '#' + s.idx + ' ' + esc(s.customer || '') +
      (s.estimated_arrival ? '<br>ETA: ' + esc(String(s.estimated_arrival)) : '')));
    if (track.length > 1) {
      const pts = track.map(p => [p.latitude, p.longitude]);
      L.polyline(pts, { color: '#2563eb', weight: 4 }).addTo(map);
      pts.forEach(p => all.push(p));
    }
    if (veh && veh.custom_last_lat && veh.custom_last_lng) {
      const p = [veh.custom_last_lat, veh.custom_last_lng];
      L.circleMarker(p, { radius: 9, weight: 2, color: '#111', fillColor: '#16a34a', fillOpacity: 0.9 })
        .addTo(map).bindPopup(esc(frm.doc.vehicle) + '<br>' + esc(String(veh.custom_last_gps_time || '')));
      all.push(p);
    }
    map.invalidateSize();
    map.fitBounds(L.latLngBounds(all), { padding: [30, 30] });
  }, 300);
};
